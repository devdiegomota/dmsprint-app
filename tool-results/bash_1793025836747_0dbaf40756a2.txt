    21→function MetricBadge({ level }: { level: "good" | "warn" | "bad" }) {
    22→  return (
    23→    <Badge
    24→      variant="secondary"
    25→      className={cn(
    26→        level === "good" && "bg-emerald-600/15 text-emerald-700 dark:text-emerald-400",
    27→        level === "warn" && "bg-amber-600/15 text-amber-700 dark:text-amber-400",
    28→        level === "bad" && "bg-red-600/15 text-red-700 dark:text-red-400",
    29→      )}
    30→    >
    31→      {level === "good" ? "OK" : level === "warn" ? "Atenção" : "Crítico"}
    32→    </Badge>
    33→  );
    34→}

    91→  onSettingsChange,
    92→  fitMode,

    95→  enhancedCanvas,

    98→  aiRestored,

   103→  sourceCanvas,
   104→  cropApplied,
   105→  cropSize,
   106→  onCropApply,
   107→  onCropRevert,
   108→  onContinue,
   109→  onResetSettings,
   110→}: {
   111→  spec: ProductSpec;
   112→  analysis: ImageAnalysis;
   113→  settings: EnhanceSettings;
   114→  onSettingsChange: (s: EnhanceSettings) => void;

   128→  cropSize: { w: number; h: number } | null;
   129→  onCropApply: (rect: CropRect) => void;

   138→  const resLevel = analysis.effectiveDpi >= 300 ? "good" : analysis.effectiveDpi >= 200 ? "warn" : "bad";

   142→  const ratioLevel = framing.ratioMatches ? "good" : framing.ratioDiff <= 0.08 ? "warn" : "bad";
   143→  const axisLabel = (axis: "x" | "y") => (axis === "x" ? "da largura" : "da altura");
   144→  const framingWarning = framing.ratioMatches
   145→    ? "Proporção da arte compatível com o produto — os três modos produzem o mesmo resultado."
   146→    : fitMode === "cover"
   147→      ? `A arte é proporcionalmente diferente do produto — no modo Preencher, ≈${framing.cropPct.toFixed(0)}% ${axisLabel(framing.cropAxis!)} serão recortados (corte central). Prefira "Conter" para mostrar a arte inteira.`
   148→      : fitMode === "contain"
   149→        ? `A arte é proporcionalmente diferente do produto — no modo Conter, surgem bordas brancas de ≈${framing.barPct.toFixed(0)}% ${axisLabel(framing.barAxis!)} em cada lado.`
   150→        : `A arte será esticada ≈${framing.stretchPct.toFixed(0)}% ${axisLabel(framing.stretchAxis!)} — textos e logotipos podem ficar visivelmente deformados. Prefira "Preencher" ou "Conter".`;
   151→
   152→  const aiRecommended = analysis.isBlurry || analysis.isLowRes;
   153→
   154→  return (
   155→    <div className="grid gap-6 lg:grid-cols-inmax(0,5fr)_minmax(0,4fr)]">
   156→      {/* Coluna esquerda: análise + controles */}
   157→      <div className="flex flex-col gap-6">
   158→        <Card>
   159→          <CardHeader>
   160→            <CardTitle>Diagnóstico da arte</CardTitle>
   161→            <CardDescription>
   162→              {spec.name} · destino {fmtMm(spec.widthMm)} × {fmtMm(spec.heightMm)} mm @ 300 DPI — fonte:{" "}
   163→              {analysis.naturalWidth} × {analysis.naturalHeight} px ({analysis.effectiveDpi} DPI efetivos)
   164→            </CardDescription>
   165→          </CardHeader>
   166→          <CardContent className="grid gap-3 sm:grid-cols-2">
   167→            <MetricCard
   168→              icon={<Gauge className="h-5 w-5" aria-hidden />}
   169→              title="Nitidez"
   170→              value={`${analysis.sharpnessScore}/100`}
   171→              detail={

   173→                  ? "Imagem com baixa nitidez — máscara de contraste aplicada automaticamente."

   175→              }
   176→              level={sharpLevel}
   177→            />
   178→            <MetricCard
   179→              icon={<ScanLine className="h-5 w-5" aria-hidden />}
   180→              title="Resolução efetiva"
   181→              value={`${analysis.effectiveDpi} DPI`}
   182→              detail={
   183→                analysis.isLowRes
   184→                  ? "Menor que 300 DPI — a arte foi reamostrada e realçada para o tamanho de corte."
   185→                  : "Resolução suficiente para impressão offset/digital sem perda."
   186→              }
   187→              level={resLevel}
   188→            />
   189→            <MetricCard
   190→              icon={<Type className="h-5 w-5" aria-hidden />}
   191→              title="Texto fino detectado"
   192→              value={`${analysis.thinTextScore}%`}
   193→              detail={
   194→                analysis.likelyThinText
   195→                  ? "Traços finos prováveis — reforço morfológico recomendado para não sumirem na impressão."

   197→              }
   198→              level={textLevel}
   199→            />
   200→            <MetricCard
   201→              icon={<Contrast className="h-5 w-5" aria-hidden />}
   202→              title="Fundo predominante"
   203→              value={analysis.lightBackground ? "Claro" : "Escuro"}
   204→              detail={
   205→                analysis.lightBackground
   206→                  ? "Reforço usará dilação do escuro (engrossa texto preto/escuro)."
   207→                  : "Reforço usará dilação do claro (engrossa texto branco/claro)."
   208→              }
   209→              level="good"
   210→            />
   211→            <MetricCard
   212→              className="sm:col-span-2"
   213→              icon={<Frame className="h-5 w-5" aria-hidden />}
   214→              title="Proporção da arte"
   215→              value={`${framing.srcRatio.toFixed(2)} : 1 · destino ${framing.targetRatio.toFixed(2)} : 1`}
   216→              detail={
   217→                framing.ratioMatches
   218→                  ? "Compatível com o tamanho de corte — sem recorte nem margens extras."
   219→                  : `Difere do corte ${fmtMm(spec.widthMm)} × ${fmtMm(spec.heightMm)} mm — escolha o enquadramento abaixo para evitar distorção.`
   220→              }
   221→              level={ratioLevel}
   222→            />
   223→          </CardContent>
   224→        </Card>
   225→
   226→        {/* Recorte da arte */}
   227→        <Card>
   228→          <CardHeader className="flex-row items-start justify-between space-y-0">
   229→            <div className="min-w-0">
   230→              <CardTitle className="flex items-center gap-2">
   231→                <Scissors className="h-5 w-5 text-amber-600" aria-hidden />
   232→                Recorte da arte
   233→              </CardTitle>
   234→              <CardDescription>
   235→                A arte veio com fundo ao redor (foto de ambiente, mockup, mesa)? Recorte exatamente
   236→                na borda do {spec.name.toLowerCase()} aqui dentro — sem precisar do Canva.
   237→              </CardDescription>
   238→            </div>
   239→            {cropApplied && (
   240→              <Badge className="shrink-0 bg-emerald-600 text-white hover:bg-emerald-600">
   241→                Recortada
   242→              </Badge>
   243→            )}
   244→          </CardHeader>
   245→          <CardContent className="flex flex-col gap-4">
   246→            {cropApplied && cropSize && (
   247→              <p className="flex items-start gap-2 rounded-lg border border-emerald-600/40 bg-emerald-600/10 p-3 text-xs leading-relaxed text-emerald-700 dark:text-emerald-400">
   248→                <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" aria-hidden />
   249→                <span>
   250→                  Recorte aplicado — a arte de trabalho agora tem {cropSize.w} × {cropSize.h} px e
   251→                  o diagnóstico acima já reflete o resultado. Repita o recorte se precisar ajustar.
   252→                </span>
   253→              </p>
   254→            )}
   255→            <p className="text-xs leading-relaxed text-muted-foreground">
   256→              1. Abra o recorte · 2. Toque em <strong className="font-semibold">Detectar bordas</strong> ou arraste
   257→              as alças até a borda da peça · 3. Aplique. Faça o recorte ANTES da restauração com IA —
   258→              assim a IA gasta resolução só na arte, não no fundo.
   259→            </p>
   260→            <div className="flex flex-wrap items-center gap-3">
   261→              <Button
   262→                variant={cropApplied ? "outline" : "default"}
   263→                className={cn(!cropApplied && "bg-amber-600 text-white hover:bg-amber-700")}
   264→                onClick={() => setCropOpen(true)}
   265→                disabled={aiRestoring}
   266→              >
   267→                <Crop className="h-4 w-4" aria-hidden />
   268→                {cropApplied ? "Recortar novamente" : "Recortar arte"}
   269→              </Button>
   270→              {cropApplied && (
   271→                <Button variant="outline" onClick={onCropRevert} disabled={aiRestoring}>
   272→                  <Undo2 className="h-4 w-4" aria-hidden />
   273→                  Desfazer recorte
   274→                </Button>
   275→              )}
   276→            </div>
   277→            <CropDialog
   278→              open={cropOpen}
   279→              onOpenChange={setCropOpen}
   280→              spec={spec}
   281→              sourceCanvas={sourceCanvas}
   282→              onApply={onCropApply}
   283→            />
   284→          </CardContent>
   285→        </Card>
   286→
   287→        <Card>
   288→          <CardHeader className="flex-row items-start justify-between space-y-0">
   289→            <div className="min-w-0">
   290→              <CardTitle className="flex items-center gap-2">
   291→                <Sparkles className="h-5 w-5 text-amber-600" aria-hidden />
   292→                Restauração com IA
   293→              </CardTitle>
   294→              <CardDescription>
   295→                Para artes muito embaçadas ou de baixa resolução: a IA recria a imagem com bordas e
   296→                textos nítidos, mantendo o layout, as cores e todos os elementos no lugar.
   297→              </CardDescription>
   298→            </div>
   299→            {aiRecommended && (
   300→              <Badge className="shrink-0 bg-amber-600 text-white hover:bg-amber-600">
   301→                Recomendada
   302→              </Badge>
   303→            )}
   304→          </CardHeader>
   305→          <CardContent className="flex flex-col gap-4">
   306→            {aiRecommended && (
   307→              <p className="rounded-lg border border-amber-600/40 bg-amber-600/10 p-3 text-xs leading-relaxed text-amber-700 dark:text-amber-400">
   308→                <TriangleAlert className="mr-1.5 inline h-3.5 w-3.5" aria-hidden />
   309→                Esta arte tem nitidez {analysis.sharpnessScore}/100 e {analysis.effectiveDpi} DPI
   310→                efetivos — abaixo do ideal para impressão. O tratamento clássico ajuda, mas não
   311→                recria detalhes que não existem no arquivo; a IA reconstrói esses detalhes.
   312→              </p>
   313→            )}
   314→
   315→            <p className="rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-xs leading-relaxed">
   316→              <strong className="font-semibold">Importante:</strong> a IA recria a arte a partir da
   317→              imagem original. Antes de imprimir, confira TODOS os textos, números de telefone,
   318→              preços e logotipos na comparação ao lado — caracteres parecidos podem ser trocados.
   319→              Se algo ficar diferente, desfaça e siga com o tratamento clássico.
   320→            </p>
   321→
   322→            {aiError && (
   323→              <p role="alert" className="rounded-lg border border-destructive/40 bg-destructive/10 p-3 text-xs leading-relaxed text-destructive">
   324→                <TriangleAlert className="mr-1.5 inline h-3.5 w-3.5" aria-hidden />
   325→                {aiError}
   326→              </p>
   327→            )}
   328→
   329→            {aiRestored && (
   330→              <p className="flex items-start gap-2 rounded-lg border border-emerald-600/40 bg-emerald-600/10 p-3 text-xs leading-relaxed text-emerald-700 dark:text-emerald-400">
   331→                <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" aria-hidden />
   332→                <span>
   333→                  Arte restaurada — o diagnóstico e o tratamento abaixo já usam a versão da IA.
   334→                  Confira os textos na comparação ao lado antes de continuar.
   335→                </span>
   336→              </p>
   337→            )}
   338→
   339→            <div className="flex flex-wrap items-center gap-3">
   340→              <Button
   341→                onClick={onAiRestore}
   342→                disabled={aiRestoring || processing}
   343→                className="bg-amber-600 text-white hover:bg-amber-700"
   344→              >
   345→                {aiRestoring ? (
   346→                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   347→                ) : (
   348→                  <Sparkles className="h-4 w-4" aria-hidden />
   349→                )}
   350→                {aiRestoring ? "Restaurando… (10–30 s)" : aiRestored ? "Restaurar novamente" : "Restaurar com IA"}
   351→              </Button>
   352→              {aiRestored && (
   353→                <Button variant="outline" onClick={onAiRevert} disabled={aiRestoring}>
   354→                  <Undo2 className="h-4 w-4" aria-hidden />
   355→                  Desfazer (voltar ao original)
   356→                </Button>
   357→              )}
   358→            </div>
   359→            <p className="text-xs leading-relaxed text-muted-foreground">
   360→              A restauração usa sempre o arquivo original como base e leva de 10 a 30 segundos.
   361→              Depois dela, todos os ajustes de tratamento continuam funcionando por cima do
   362→              resultado.
   363→            </p>
   364→          </CardContent>
   365→        </Card>
   366→
   367→        <Card>
   368→          <CardHeader className="flex-row items-start justify-between space-y-0">
   369→            <div>
   370→              <CardTitle>Tratamento aplicado</CardTitle>
   371→              <CardDescription>Ajuste fino em tempo real — a pré-visualização ao lado atualiza automaticamente.</CardDescription>
   372→            </div>
   373→            <Button variant="outline" size="sm" onClick={onResetSettings}>
   374→              <RotateCcw className="h-4 w-4" aria-hidden />
   375→              Automático
   376→            </Button>
   377→          </CardHeader>
   378→          <CardContent className="flex flex-col gap-5">
   379→            {/* Enquadramento (geometria — nunca distorce em cover/contain) */}
   380→            <div className="rounded-lg border p-4">
   381→              <Label className="text-sm font-semibold">Enquadramento no tamanho de corte</Label>
   382→              <p className="mt-1 text-xs leading-relaxed text-muted-foreground">
   383→                Define como a arte ocupa o corte de {fmtMm(spec.widthMm)} × {fmtMm(spec.heightMm)} mm. Os modos
   384→                "Preencher" e "Conter" preservam a proporção original — a imagem nunca fica esticada.
   385→              </p>
   386→              <RadioGroup
   387→                value={fitMode}
   388→                onValueChange={(v) => onFitModeChange(v as FitMode)}
   389→                className="mt-3 grid gap-2"
   390→                aria-label="Modo de enquadramento"

   396→                    className={cn(
   397→                      "flex cursor-pointer items-start gap-3 rounded-lg border p-3 transition-colors font-normal",
   398→                      fitMode === opt.value
   399→                        ? "border-amber-600 bg-amber-600/10 ring-1 ring-amber-600"
   400→                        : "hover:bg-muted/60",
   401→                    )}
   402→                  >
   403→                    <RadioGroupItem id={`fit-${opt.value}`} value={opt.value} className="mt-0.5" />
   404→                    <span className="min-w-0 flex-1">
   405→                      <span className="flex items-center gap-1.5 text-sm font-semibold">
   406→                        {opt.icon}

   408→                      </span>

   414→              <p

   416→                className={cn(
   417→                  "mt-3 rounded-lg border p-3 text-xs leading-relaxed",
   418→                  framing.ratioMatches
   419→                    ? "border-border bg-muted/60 text-muted-foreground"
   420→                    : fitMode === "stretch"
   421→                      ? "border-destructive/40 bg-destructive/10 text-destructive"
   422→                      : "border-amber-600/40 bg-amber-600/10 text-amber-700 dark:text-amber-400",
   423→                )}
   424→              >
   425→                {!framing.ratioMatches && <TriangleAlert className="mr-1.5 inline h-3.5 w-3.5" aria-hidden />}
   426→                {framingWarning}
   427→              </p>
   428→            </div>
   429→
   430→            {/* Nitidez */}
   431→            <div className="rounded-lg border p-4">
   432→              <div className="flex items-center justify-between">
   433→                <Label htmlFor="sw-sharpen" className="text-sm font-semibold">

   450→                    value={[settings.sharpenAmount]}
