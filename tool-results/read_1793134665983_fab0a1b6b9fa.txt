    37→    37→/** Limita o deslocamento para a imagem nunca sair totalmente da área visível. */
    38→    38→function clampPan(
    39→    39→  p: { x: number; y: number },
    40→    40→  dispW: number,
    41→    41→  dispH: number,
    42→    42→  stageW: number,
    43→    43→  stageH: number,
    44→    44→) {
    45→    45→  const mx = Math.max(0, (dispW - stageW) / 2);
    46→    46→  const my = Math.max(0, (dispH - stageH) / 2);
    47→    47→  return { x: clampNum(p.x, -mx, mx), y: clampNum(p.y, -my, my) };
    48→    48→}

    83→    83→ * Diálogo de recorte interativo: quadro arrastável com 8 alças sobre a arte,

    92→    92→export function CropDialog({
    93→    93→  open,
    94→    94→  onOpenChange,
    95→    95→  spec,
    96→    96→  sourceCanvas,
    97→    97→  onApply,
    98→    98→}: {
    99→    99→  open: boolean;
   100→   100→  onOpenChange: (v: boolean) => void;
   101→   101→  spec: ProductSpec;
   102→   102→  sourceCanvas: HTMLCanvasElement | null;
   103→   103→  onApply: (rect: CropRect, rotationQuarters: number) => void;
   104→   104→}) {
   105→   105→  const [rect, setRect] = useState<CropRect | null>(null);
   106→   106→  const [ratioId, setRatioId] = useState("product");
   107→   107→  const [detecting, setDetecting] = useState(false);
   108→   108→  const [detectHint, setDetectHint] = useState<string | null>(null);
   109→   109→  const [dataUrl, setDataUrl] = useState<string | null>(null);
   110→   110→  const [dragging, setDragging] = useState(false);
   111→   111→  /** Giro aplicado à arte dentro do diálogo, em quartos de volta horários (0–3). */
   112→   112→  const [rot, setRot] = useState(0);
   113→   113→
   114→   114→  // Zoom e navegação da imagem
   115→   115→  const [zoom, setZoom] = useState(1);
   116→   116→  const [pan, setPan] = useState({ x: 0, y: 0 });
   117→   117→  const [hand, setHand] = useState(false);
   118→   118→  const [panning, setPanning] = useState(false);
   119→   119→  const [stage, setStage] = useState({ w: 0, h: 0 });
   120→   120→
   121→   121→  const stageRef = useRef<HTMLDivElement | null>(null);
   122→   122→  const stageRoRef = useRef<ResizeObserver | null>(null);
   123→   123→  const wrapRef = useRef<HTMLDivElement | null>(null);
   124→   124→  const dragRef = useRef<DragState | null>(null);
   125→   125→  const panRef = useRef<{ startX: number; startY: number; basePan: { x: number; y: number } } | null>(
   126→   126→    null,
   127→   127→  );
   128→   128→
   129→   129→  const presets = useMemo(() => cropPresets(spec), [spec]);
   130→   130→  const ratio = presets.find((p) => p.id === ratioId)?.ratio ?? null;
   131→   131→
   132→   132→  // Canvas de trabalho = fonte já girada pelos quartos de volta escolhidos.
   133→   133→  // Todo o diálogo (preview, detecção, quadro) opera sobre ESTE canvas; o giro
   134→   134→  // só vira parte do pipeline quando o usuário confirma "Aplicar".
   135→   135→  const working = useMemo(
   136→   136→    () => (sourceCanvas && rot !== 0 ? rotateCanvas(sourceCanvas, rot) : sourceCanvas),
   137→   137→    [sourceCanvas, rot],
   138→   138→  );
   139→   139→
   140→   140→  const W = working?.width ?? 0;
   141→   141→  const H = working?.height ?? 0;
   142→   142→
   143→   143→  // Preview em JPEG leve (a fonte pode ter vários megapixels).
   144→   144→  useEffect(() => {
   145→   145→    if (!open || !working) {
   146→   146→      setDataUrl(null);
   147→   147→      return;
   148→   148→    }
   149→   149→    try {
   150→   150→      setDataUrl(working.toDataURL("image/jpeg", 0.9));
   151→   151→    } catch {
   152→   152→      setDataUrl(null);
   153→   153→    }
   154→   154→  }, [open, working]);
   155→   155→
   156→   156→  // Ao abrir: trava no produto, quadro centralizado e visualização redefinida.
   157→   157→  // Roda UMA VEZ por abertura (transição fechado→aberto) — NÃO depende de W/H,
   158→   158→  // senão um giro dentro do diálogo reencaixaria o quadro por conta própria e
   159→   159→  // desfaria a transformação feita por setRotation.
   160→   160→  const wasOpenRef = useRef(false);
   161→   161→  useEffect(() => {
   162→   162→    if (open && !wasOpenRef.current && sourceCanvas) {
   163→   163→      wasOpenRef.current = true;
   164→   164→      setRot(0);
   165→   165→      setRatioId("product");
   166→   166→      setRect(
   167→   167→        fitRectWithRatio(
   168→   168→          sourceCanvas.width,
   169→   169→          sourceCanvas.height,
   170→   170→          spec.widthMm / spec.heightMm,
   171→   171→        ),
   172→   172→      );
   173→   173→      setDetectHint(null);
   174→   174→      setDetecting(false);
   175→   175→      setZoom(1);
   176→   176→      setPan({ x: 0, y: 0 });
   177→   177→      setHand(false);
   178→   178→    }
   179→   179→    if (!open) wasOpenRef.current = false;
   180→   180→  }, [open, sourceCanvas, spec]);
   181→   181→
   182→   182→  // Mede a área da imagem via CALLBACK REF — o conteúdo do diálogo (portal Radix)
   183→   183→  // monta depois do primeiro passe de efeitos, então um useEffect [open] via cedo
   184→   184→  // e nunca re-tenta. O callback ref roda exatamente na montagem do elemento.
   185→   185→  const attachStage = useCallback((el: HTMLDivElement | null) => {
   186→   186→    stageRef.current = el;
   187→   187→    stageRoRef.current?.disconnect();
   188→   188→    stageRoRef.current = null;
   189→   189→    if (el) {
   190→   190→      const measure = () => setStage({ w: el.clientWidth, h: el.clientHeight });
   191→   191→      measure();
   192→   192→      const ro = new ResizeObserver(measure);
   193→   193→      ro.observe(el);
   194→   194→      stageRoRef.current = ro;
   195→   195→    }
   196→   196→  }, []);
   197→   197→
   198→   198→  // Escala "ajustada" (imagem inteira visível) e tamanho exibido com o zoom atual.
   199→   199→  const fitScale =
   200→   200→    stage.w > 48 && stage.h > 48 && W > 0 && H > 0
   201→   201→      ? Math.min((stage.w - 32) / W, (stage.h - 32) / H)
   202→   202→      : 0;
   203→   203→  const dispW = fitScale > 0 ? W * fitScale * zoom : 0;
   204→   204→  const dispH = fitScale > 0 ? H * fitScale * zoom : 0;
   205→   205→
   206→   206→  const panClamped = useCallback(
   207→   207→    (p: { x: number; y: number }, z: number) =>
   208→   208→      clampPan(p, W * fitScale * z, H * fitScale * z, stage.w, stage.h),
   209→   209→    [W, H, fitScale, stage],
   210→   210→  );
   211→   211→
   212→   212→  const applyZoom = useCallback(
   213→   213→    (target: number, anchor?: { cx: number; cy: number }) => {
   214→   214→      const next = clampNum(target, 1, MAX_ZOOM);
   215→   215→      if (!Number.isFinite(next) || next === zoom || fitScale === 0) return;
   216→   216→      if (anchor) {
   217→   217→        const k = next / zoom;
   218→   218→        setPan(
   219→   219→          panClamped(
   220→   220→            { x: anchor.cx - (anchor.cx - pan.x) * k, y: anchor.cy - (anchor.cy - pan.y) * k },
   221→   221→            next,
   222→   222→          ),
   223→   223→        );
   224→   224→      } else {
   225→   225→        setPan((p) => panClamped(p, next));
   226→   226→      }
   227→   227→      setZoom(next);
   228→   228→    },
   229→   229→    [zoom, pan, fitScale, panClamped],
   230→   230→  );
   231→   231→
   232→   232→  /**
   233→   233→   * Aplica `next` quartos de volta (horários) à vista. O quadro acompanha:
   234→   234→   * com trava de proporção, reencaixa na proporção travada na nova direção;
   235→   235→   * em modo livre, a MESMA região selecionada é transformada junto com a imagem.
   236→   236→   */
   237→   237→  const setRotation = (next: number) => {
   238→   238→    if (!rect || W === 0 || H === 0) return;
   239→   239→    const delta = (((next - rot) % 4) + 4) % 4;
   240→   240→    if (delta === 0) return;
   241→   241→    const nw = delta % 2 === 1 ? H : W;
   242→   242→    const nh = delta % 2 === 1 ? W : H;
   243→   243→    if (ratio) {
   244→   244→      setRect(fitRectWithRatio(nw, nh, ratio));
   245→   245→    } else {
   246→   246→      setRect(clampRect(rotateRect(rect, delta, W, H), nw, nh));
   247→   247→    }
   248→   248→    setRot(((next % 4) + 4) % 4);
   249→   249→    setZoom(1);
   250→   250→    setPan({ x: 0, y: 0 });
   251→   251→    setDetectHint(null);
   252→   252→  };
   253→   253→
   254→   254→  const handleRatioChange = (id: string) => {
   255→   255→    setRatioId(id);
   256→   256→    const r = presets.find((p) => p.id === id)?.ratio ?? null;
   257→   257→    if (r && W > 0 && H > 0) setRect(fitRectWithRatio(W, H, r));
   258→   258→  };
   259→   259→
   260→   260→  const handleDetect = () => {
   261→   261→    if (!working || detecting) return;
   262→   262→    setDetecting(true);
   263→   263→    setDetectHint(null);
   264→   264→    // Pequeno atraso para o spinner pintar antes do cálculo síncrono.
   265→   265→    setTimeout(() => {
   266→   266→      try {
   267→   267→        const found = detectArtworkBounds(working);
   268→   268→        if (found) {
   269→   269→          setRect(found);
   270→   270→          setRatioId("free"); // a detecção reflete a proporção real da peça
   271→   271→          setDetectHint(null);
   272→   272→        } else {
   273→   273→          setDetectHint(
   274→   274→            "Não consegui identificar a borda com confiança — arraste as alças até a borda da peça.",
   275→   275→          );
   276→   276→        }
   277→   277→      } finally {
   278→   278→        setDetecting(false);
   279→   279→      }
   280→   280→    }, 40);
   281→   281→  };
   282→   282→
   283→   283→  // Converte coordenadas de tela para pixels da imagem (funciona com zoom e pan:
   284→   284→  // getBoundingClientRect reflete a posição/tamanho visuais do quadro).
   285→   285→  const toImage = (clientX: number, clientY: number) => {
   286→   286→    const el = wrapRef.current;
   287→   287→    if (!el || W === 0 || H === 0) return null;
   288→   288→    const r = el.getBoundingClientRect();
   289→   289→    return { x: ((clientX - r.left) / r.width) * W, y: ((clientY - r.top) / r.height) * H };
   290→   290→  };
   291→   291→
   292→   292→  const startDrag = (handle: CropHandle) => (e: React.PointerEvent) => {
   293→   293→    if (!rect) return;
   294→   294→    e.preventDefault();
   295→   295→    e.stopPropagation();
   296→   296→    const p = toImage(e.clientX, e.clientY);
   297→   297→    if (!p) return;
   298→   298→    dragRef.current = { handle, startRect: rect, startX: p.x, startY: p.y };
   299→   299→    setDragging(true);
   300→   300→  };
   301→   301→
   302→   302→  const startPan = (e: React.PointerEvent) => {
   303→   303→    e.preventDefault();
   304→   304→    e.stopPropagation();
   305→   305→    panRef.current = { startX: e.clientX, startY: e.clientY, basePan: pan };
   306→   306→    setPanning(true);
   307→   307→  };
   308→   308→
   309→   309→  // Clique na imagem: dentro do quadro (ou zoom 100%) move o quadro;
   310→   310→  // fora do quadro com zoom, ou no modo mãozinha, navega pela imagem.
   311→   311→  const onImagePointerDown = (e: React.PointerEvent) => {
   312→   312→    if (!rect) return;
   313→   313→    const p = toImage(e.clientX, e.clientY);
   314→   314→    const inside =
   315→   315→      p && p.x >= rect.x && p.x <= rect.x + rect.width && p.y >= rect.y && p.y <= rect.y + rect.height;
   316→   316→    if (hand || (!inside && zoom > 1)) {
   317→   317→      startPan(e);
   318→   318→    } else {
   319→   319→      startDrag("move")(e);
   320→   320→    }
   321→   321→  };
   322→   322→
   323→   323→  // Loop de arrasto do quadro em nível de janela (sobrevive a sair da imagem).
   324→   324→  useEffect(() => {
   325→   325→    if (!dragging) return;
   326→   326→    const onMove = (ev: PointerEvent) => {
   327→   327→      const d = dragRef.current;
   328→   328→      if (!d) return;
   329→   329→      const p = toImage(ev.clientX, ev.clientY);
   330→   330→      if (!p) return;
   331→   331→      if (d.handle === "move") {
   332→   332→        setRect(moveRect(d.startRect, p.x - d.startX, p.y - d.startY, W, H));
   333→   333→      } else {
   334→   334→        setRect(resizeRect(d.startRect, d.handle, p.x, p.y, ratio, W, H));
   335→   335→      }
   336→   336→    };
   337→   337→    const onUp = () => {
   338→   338→      dragRef.current = null;
   339→   339→      setDragging(false);
   340→   340→    };
   341→   341→    window.addEventListener("pointermove", onMove);
   342→   342→    window.addEventListener("pointerup", onUp);
   343→   343→    window.addEventListener("pointercancel", onUp);
   344→   344→    return () => {
   345→   345→      window.removeEventListener("pointermove", onMove);
   346→   346→      window.removeEventListener("pointerup", onUp);
   347→   347→      window.removeEventListener("pointercancel", onUp);
   348→   348→    };
   349→   349→  }, [dragging, ratio, W, H]);
   350→   350→
   351→   351→  // Loop de arrasto do pan (navegação da imagem ampliada).
   352→   352→  useEffect(() => {
   353→   353→    if (!panning) return;
   354→   354→    const onMove = (ev: PointerEvent) => {
   355→   355→      const d = panRef.current;
   356→   356→      if (!d) return;
   357→   357→      setPan(panClamped({ x: d.basePan.x + (ev.clientX - d.startX), y: d.basePan.y + (ev.clientY - d.startY) }, zoom));
   358→   358→    };
   359→   359→    const onUp = () => {
   360→   360→      panRef.current = null;
   361→   361→      setPanning(false);
   362→   362→    };
   363→   363→    window.addEventListener("pointermove", onMove);
   364→   364→    window.addEventListener("pointerup", onUp);
   365→   365→    window.addEventListener("pointercancel", onUp);
   366→   366→    return () => {
   367→   367→      window.removeEventListener("pointermove", onMove);
   368→   368→      window.removeEventListener("pointerup", onUp);
   369→   369→      window.removeEventListener("pointercancel", onUp);
   370→   370→    };
   371→   371→  }, [panning, zoom, panClamped]);
   372→   372→
   373→   373→  // Ctrl + roda do mouse (ou pinça no trackpad): zoom centrado no cursor.
   374→   374→  useEffect(() => {
   375→   375→    const el = stageRef.current;
   376→   376→    if (!el || !open || fitScale === 0) return;
   377→   377→    const onWheel = (e: WheelEvent) => {
   378→   378→      if (!e.ctrlKey) return;
   379→   379→      e.preventDefault();
   380→   380→      const r = el.getBoundingClientRect();
   381→   381→      applyZoom(zoom * (e.deltaY < 0 ? 1.2 : 1 / 1.2), {
   382→   382→        cx: e.clientX - (r.left + r.width / 2),
   383→   383→        cy: e.clientY - (r.top + r.height / 2),
   384→   384→      });
   385→   385→    };
   386→   386→    el.addEventListener("wheel", onWheel, { passive: false });
   387→   387→    return () => el.removeEventListener("wheel", onWheel);
   388→   388→  }, [open, fitScale, zoom, applyZoom]);
   389→   389→
   390→   390→  // Mantém o pan válido quando a área/zoom muda (ex.: redimensionar a janela).
   391→   391→  useEffect(() => {
   392→   392→    setPan((p) => panClamped(p, zoom));
   393→   393→  }, [stage, fitScale, zoom, panClamped]);
   394→   394→
   395→   395→  // Setas do teclado: move o quadro em 1 px (Shift = 10 px).
   396→   396→  useEffect(() => {
   397→   397→    if (!open) return;
   398→   398→    const onKey = (e: KeyboardEvent) => {
   399→   399→      const target = e.target as HTMLElement | null;
   400→   400→      if (target?.closest?.('[role="listbox"], [role="option"], [role="menu"]')) return;
   401→   401→      const step = e.shiftKey ? 10 : 1;
   402→   402→      const delta =
   403→   403→        e.key === "ArrowLeft"
   404→   404→          ? [-step, 0]
   405→   405→          : e.key === "ArrowRight"
   406→   406→            ? [step, 0]
   407→   407→            : e.key === "ArrowUp"
   408→   408→              ? [0, -step]
   409→   409→              : e.key === "ArrowDown"
   410→   410→                ? [0, step]
   411→   411→                : null;
   412→   412→      if (!delta) return;
   413→   413→      e.preventDefault();
   414→   414→      setRect((r) => (r ? moveRect(r, delta[0], delta[1], W, H) : r));
   415→   415→    };
   416→   416→    window.addEventListener("keydown", onKey);
   417→   417→    return () => window.removeEventListener("keydown", onKey);
   418→   418→  }, [open, W, H]);
   419→   419→
   420→   420→  const handleApply = () => {
   421→   421→    if (!rect || W === 0 || H === 0) return;
   422→   422→    onApply(clampRect(rect, W, H), rot);
   423→   423→    onOpenChange(false);
   424→   424→  };
   425→   425→
   426→   426→  const dpi = rect ? cropEffectiveDpi(rect, spec) : null;
   427→   427→  const dpiColor =
